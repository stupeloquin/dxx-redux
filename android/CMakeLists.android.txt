cmake_minimum_required(VERSION 3.28)
project(dxx-redux-android C CXX)

# DXX_GAME is passed from build.gradle: "d1x" or "d2x"
if(NOT DXX_GAME)
    set(DXX_GAME "d1x")
endif()

if(DXX_GAME STREQUAL "d2x")
    set(DXX_SRCDIR "d2")
else()
    set(DXX_SRCDIR "d1")
endif()

# Root of the dxx-redux source tree
set(DXX_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../../../../dxx-redux/${DXX_SRCDIR}")

# --------------------------------------------------------------------------
# Build SDL2 from source
# --------------------------------------------------------------------------
set(SDL2_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../SDL")
add_subdirectory(${SDL2_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/SDL2-build)

# --------------------------------------------------------------------------
# Build SDL2_mixer from source
# --------------------------------------------------------------------------
set(SDL2_MIXER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../SDL2_mixer")
set(SDL2_DIR "${CMAKE_CURRENT_BINARY_DIR}/SDL2-build")
set(SDL2_INCLUDE_DIR "${SDL2_SOURCE_DIR}/include")

set(SDL2MIXER_OPUS OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_FLAC OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_MOD OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_MIDI OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_WAVPACK OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_MP3 OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_OGG OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_INSTALL OFF CACHE BOOL "" FORCE)

add_subdirectory(${SDL2_MIXER_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/SDL2_mixer-build)

# --------------------------------------------------------------------------
# PhysFS (pre-built static library)
# --------------------------------------------------------------------------
set(PHYSFS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../physfs-install")
add_library(physfs STATIC IMPORTED)
set_target_properties(physfs PROPERTIES
    IMPORTED_LOCATION "${PHYSFS_ROOT}/lib/libphysfs.a"
)
set(PHYSFS_INCLUDE_DIR "${PHYSFS_ROOT}/include")

# Game-specific sources
# --------------------------------------------------------------------------
if(DXX_GAME STREQUAL "d2x")

    list(APPEND SRC_MAIN
        ${DXX_SRC}/main/ai2.c
        ${DXX_SRC}/main/escort.c
        ${DXX_SRC}/main/gamepal.c
        ${DXX_SRC}/main/movie.c

    set(SRC_LIBMVE

        ${DXX_SRC}/libmve/mve_audio.c
        ${DXX_SRC}/libmve/mvelib.c

    # D1-specific sources
    list(APPEND SRC_MAIN
        ${DXX_SRC}/main/bmread.c
        ${DXX_SRC}/main/custom.c
        ${DXX_SRC}/main/snddecom.c
    )
    set(SRC_LIBMVE "")
endif()

# --------------------------------------------------------------------------
# xmodel stub (xmodel requires GLEW/GL2.0+, not available on GLES 1.x)
# --------------------------------------------------------------------------
set(SRC_XMODEL_STUB
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../../dxx-redux/android/xmodel_stub.c
)

# --------------------------------------------------------------------------
# Build as shared library (libmain.so) for SDL2 Android
# --------------------------------------------------------------------------
add_library(main SHARED
    ${SRC_2D}
    ${SRC_3D}
    ${SRC_IFF}
    ${SRC_MATHS}
    ${SRC_MEM}
    ${SRC_MISC}
    ${SRC_TEXMAP}
    ${SRC_ARCH_OGL}
    ${SRC_ARCH_SDL}
    ${SRC_MAIN}
    ${SRC_LIBMVE}
    ${SRC_XMODEL_STUB}
)

# --------------------------------------------------------------------------
# Include directories
# --------------------------------------------------------------------------
target_include_directories(main PRIVATE
    ${DXX_SRC}/include
    ${DXX_SRC}/arch/include
    ${DXX_SRC}/main
    ${SDL2_SOURCE_DIR}/include
    ${SDL2_MIXER_SOURCE_DIR}/include
    ${PHYSFS_INCLUDE_DIR}
)

# --------------------------------------------------------------------------
# Compile definitions
# --------------------------------------------------------------------------
target_compile_definitions(main PRIVATE
    # Platform
    OGL
    OGLES
    NETWORK
    USE_SDLMIXER
    USE_UDP
    USE_TRACKER
    NO_ASM
    HAVE_STRUCT_TIMESPEC
    HAVE_STRUCT_TIMEVAL
    SHAREPATH=""
    # Version info
    DXX_VERSION_MAJORi=1
    DXX_VERSION_MINORi=1
    DXX_VERSION_MICROi=0
)

# --------------------------------------------------------------------------
# Compile options
# --------------------------------------------------------------------------
target_compile_options(main PRIVATE
    -Wno-address-of-packed-member
    -Wno-deprecated-declarations
    -Wno-format-truncation
    -Wno-format-overflow
)

# --------------------------------------------------------------------------
# Link libraries
# --------------------------------------------------------------------------
target_link_libraries(main PRIVATE
    SDL2
    SDL2_mixer
    physfs
    GLESv1_CM
    EGL
    log
    android
    dl
    m
)
